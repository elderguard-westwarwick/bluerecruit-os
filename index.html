Copy the Web App URL into intake.html // ============================================================ // ── PLACEHOLDERS — update these before deploying ────────────── var SHEET_ID = 'YOUR_GOOGLE_SHEET_ID_HERE'; // How to find: open your Google Sheet, copy the ID from the URL // e.g. docs.google.com/spreadsheets/d/THIS_PART_HERE/edit var NOTIFICATION_EMAIL = 'YOUR_EMAIL_ADDRESS_HERE'; // e.g. 'you@gmail.com' or 'admin@bluerecruit.com' var SHEET_NAME = 'Submissions'; // Name of the tab in your Google Sheet // ────────────────────────────────────────────────────────────── // ── doPost: receives form submission ────────────────────────── function doPost(e) { try { var data = JSON.parse(e.postData.contents); writeToSheet(data); sendNotification(data); return ContentService .createTextOutput(JSON.stringify({ status: 'success' })) .setMimeType(ContentService.MimeType.JSON); } catch (err) { return ContentService .createTextOutput(JSON.stringify({ status: 'error', message: err.message })) .setMimeType(ContentService.MimeType.JSON); } } // ── doGet: simple health check ──────────────────────────────── function doGet(e) { return ContentService .createTextOutput('BlueRecruit Intake API is live.') .setMimeType(ContentService.MimeType.TEXT); } // ── writeToSheet: logs submission as a new row ──────────────── function writeToSheet(data) { var ss = SpreadsheetApp.openById(SHEET_ID); var sheet = ss.getSheetByName(SHEET_NAME); // Create header row if sheet is empty if (sheet.getLastRow() === 0) { sheet.appendRow([ 'Timestamp', 'Department Name', 'City', 'County', 'State', 'Agency Type', 'Authorized Strength', 'Current Officers', 'Vacancies', 'Civilian Staff', 'Hiring Timeline', 'Priority Hire Type', 'Contact Name', 'Contact Title', 'Contact Email', 'Contact Phone', 'Preferred Contact Method', 'Best Contact Time', 'Pain Points', 'Current Channels', 'What Worked', 'What Failed', 'Recruiting Budget', 'Has Grants', 'Current Grants', 'Open to Funding', 'Campaign Channels', 'Last Recruit Effort', 'Campaign Start', 'Additional Notes' ]); // Style header row var headerRange = sheet.getRange(1, 1, 1, 30); headerRange.setBackground('#0d1b2a'); headerRange.setFontColor('#c9a84c'); headerRange.setFontWeight('bold'); sheet.setFrozenRows(1); } // Append the submission as a new row sheet.appendRow([ new Date().toLocaleString(), data.dept_name || '', data.city || '', data.county || '', data.state || '', data.agency_type || '', data.authorized_strength || '', data.current_officers || '', data.vacancies || '', data.civilian_staff || '', data.hiring_timeline || '', data.hire_type || '', data.contact_name || '', data.contact_title || '', data.contact_email || '', data.contact_phone || '', data.contact_method || '', data.contact_time || '', Array.isArray(data.pain_points) ? data.pain_points.join(', ') : (data.pain_points || ''), Array.isArray(data.current_channels) ? data.current_channels.join(', ') : (data.current_channels || ''), data.what_worked || '', data.what_failed || '', data.recruiting_budget || '', data.has_grants || '', data.current_grants || '', data.open_to_funding || '', Array.isArray(data.campaign_channels) ? data.campaign_channels.join(', ') : (data.campaign_channels || ''), data.last_recruit_effort || '', data.campaign_start || '', data.additional_notes || '' ]); } // ── sendNotification: emails you on new submission ──────────── function sendNotification(data) { var subject = '🚔 New BlueRecruit Intake — ' + (data.dept_name || 'Unknown Department'); var body = [ 'A new agency intake has been submitted.\n', '────────────────────────────────', 'DEPARTMENT: ' + (data.dept_name || ''), 'LOCATION: ' + [data.city, data.state].filter(Boolean).join(', '), 'AGENCY TYPE: ' + (data.agency_type || ''), '────────────────────────────────', 'CONTACT: ' + (data.contact_name || '') + ' — ' + (data.contact_title || ''), 'EMAIL: ' + (data.contact_email || ''), 'PHONE: ' + (data.contact_phone || ''), '────────────────────────────────', 'VACANCIES: ' + (data.vacancies || 'Not specified'), 'HIRE TYPE: ' + (data.hire_type || ''), 'TIMELINE: ' + (data.hiring_timeline || ''), '────────────────────────────────', 'PAIN POINTS: ' + (Array.isArray(data.pain_points) ? data.pain_points.join(', ') : (data.pain_points || '')), 'BUDGET: ' + (data.recruiting_budget || ''), 'OPEN TO FUNDING: ' + (data.open_to_funding || ''), '────────────────────────────────\n', 'View all submissions in Google Sheets:', 'https://docs.google.com/spreadsheets/d/' + SHEET_ID ].join('\n'); MailApp.sendEmail(NOTIFICATION_EMAIL, subject, body); }
